{{/* Preserve the precision of verified bibliographic dates. */}} {{- $date := printf "%v" . -}} {{- $display := $date -}} {{- if findRE `^[0-9]{4}-[0-9]{2}-[0-9]{2}$` $date -}} {{- $display = time.Format "2 January 2006" $date -}} {{- else if findRE `^[0-9]{4}-[0-9]{2}$` $date -}} {{- $display = time.Format "January 2006" (printf "%s-01" $date) -}} {{- end -}} {{- return $display -}}
